# src/peano_ai/models.py
from dataclasses import dataclass
from enum import Enum

class NaturalNumber:
    def __init__(self, value: int):
        if value < 0:
            raise ValueError("自然数必须非负")
        self.value = value

    @classmethod
    def zero(cls):
        return cls(0)

    def successor(self):
        return NaturalNumber(self.value + 1)

    def predecessor(self):
        if self.value == 0:
            raise ValueError("0 没有前驱")
        return NaturalNumber(self.value - 1)

    def __eq__(self, other):
        return self.value == other.value
    def __hash__(self):
        return hash(self.value)
    def __repr__(self):
        return f"Nat({self.value})"

class Term:
    def __init__(self, name: str, is_variable: bool = False):
        self.name = name
        self.is_variable = is_variable

    def __repr__(self):
        return self.name

@dataclass
class Axiom:
    name: str
    formula: str
    description: str

class PeanoAxioms:
    def __init__(self, axioms: list[Axiom]):
        self.axioms = axioms

    @classmethod
    def standard(cls):
        return cls([
            Axiom("P1", "0 ∈ ℕ", "0 是自然数"),
            Axiom("P2", "S: ℕ → ℕ", "后继函数存在"),
            Axiom("P3", "∀n: S(n) ≠ 0", "0 不是任何数的后继"),
            Axiom("P4", "∀m,n: S(m)=S(n) → m=n", "后继函数单射"),
            Axiom("P5", "∀P: P(0) ∧ ∀n: P(n)→P(S(n)) → ∀n: P(n)", "数学归纳法"),
        ])
```

# src/peano_ai/arithmetic.py
from src.peano_ai.models import Expr, Zero, Succ, Var, Add as ModelAdd, Mul as ModelMul
from typing import Optional

class DerivationEngine:
    def add(self, a: Expr, b: Expr) -> Expr:
        if isinstance(a, Zero):
            return b
        if isinstance(a, Succ):
            return Succ(self.add(a.n, b))
        if isinstance(b, Zero):
            return a
        raise NotImplementedError(f"add({a}, {b})")

    def mul(self, a: Expr, b: Expr) -> Expr:
        if isinstance(a, Zero):
            return Zero()
        if isinstance(a, Succ):
            return self.add(b, self.mul(a.n, b))
        raise NotImplementedError(f"mul({a}, {b})")

    def to_nat(self, expr: Expr) -> int:
        if isinstance(expr, Zero):
            return 0
        if isinstance(expr, Succ):
            return 1 + self.to_nat(expr.n)
        raise ValueError(f"无法将 {expr} 转换为自然数")
```

from typing import Callable, Any

@dataclass
class VerificationResult:
    is_valid: bool
    message: str

class AxiomVerifier:
    def verify_injection(self, domain, codomain) -> VerificationResult:
        if len(domain) == len(set(domain)):
            return VerificationResult(True, "函数是单射的")
        return VerificationResult(False, "函数不是单射的")

    def verify_induction(self, axioms, predicate: Callable) -> bool:
        base = predicate(0)

# src/peano_ai/modified_systems.py
from src.peano_ai.models import PeanoAxioms, Axiom

class ModifiedPeanoSystem:
    def __init__(self, axiom3: bool = True, axiom4: bool = True, axiom5: bool = True):
        self.axiom3_enabled = axiom3
        self.axiom4_enabled = axiom4
        self.axiom5_enabled = axiom5

    def verify_axiom3(self) -> bool:
        return self.axiom3_enabled

    def get_system_name(self) -> str:
        disabled = []
        if not self.axiom3_enabled:
            disabled.append("P3")
        if not self.axiom4_enabled:
            disabled.append("P4")
        if not self.axiom5_enabled:
            disabled.append("P5")
        if disabled:
            return f"Modified({','.join(disabled)})"
        return "Standard Peano"

class FiniteCyclicSystem(ModifiedPeanoSystem):
    def __init__(self, mod: int):
        super().__init__(axiom3=False)
        self.mod = mod

    def eval(self, expr: str) -> int:
        parts = expr.split("+")
        if len(parts) == 2:
            a, b = int(parts[0].strip()), int(parts[1].strip())
            return (a + b) % self.mod
        raise ValueError(f"无法解析: {expr}")

    def get_system_name(self) -> str:
        return f"Z/{self.mod}Z (模 {self.mod} 算术)"
```

# src/peano_ai/explorer.py
from src.peano_ai.arithmetic import DerivationEngine
from src.peano_ai.modified_systems import ModifiedPeanoSystem, FiniteCyclicSystem

class MathExplorer:
    def __init__(self):
        self.engine = DerivationEngine()
        self.current_system = "standard"
        self.mod_system = None

    def set_system(self, system: str):
        if system.startswith("mod"):
            mod = int(system[3:])
            self.mod_system = FiniteCyclicSystem(mod)
            self.current_system = system
        else:
            self.current_system = system

    def explore(self, expr: str):
        if self.current_system != "standard" and self.mod_system:
            return self.mod_system.eval(expr)
        parts = expr.replace(" ", "").split("+")
        if len(parts) == 2:
            a, b = int(parts[0]), int(parts[1])
            return a + b
        return int(expr)
```
